\section{Introducción}

\begin{frame}{Contexto del Proyecto}
\begin{block}{JiovaniGo - E-commerce de Perfumes de Alta Gama}
Plataforma de comercio electrónico para la venta de perfumes premium en el mercado chileno.
\end{block}

\vspace{0.5cm}

\textbf{Arquitectura:}
\begin{itemize}
    \item \textcolor{goldprimary}{\faServer} Backend: Node.js + Express
    \item \textcolor{accentblue}{\faReact} Frontend: React con gestión de estado global
    \item \textcolor{successgreen}{\faDatabase} Base de Datos: MongoDB Atlas
    \item \textcolor{goldsecondary}{\faCreditCard} Pagos: Transbank WebPay Plus
\end{itemize}
\end{frame}

\begin{frame}{Funcionalidades Principales}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{Gestión de Usuarios}
\begin{itemize}
    \item Autenticación JWT
    \item Roles (Usuario/Admin)
    \item Gestión de sesiones
\end{itemize}

\vspace{0.3cm}

\textbf{Catálogo de Productos}
\begin{itemize}
    \item Segmentación por género
    \item Búsqueda y filtros
    \item Imágenes de alta calidad
\end{itemize}
\end{column}

\begin{column}{0.48\textwidth}
\textbf{Proceso de Compra}
\begin{itemize}
    \item Carrito de compras
    \item Checkout con validación
    \item Integración Transbank
    \item Seguimiento de órdenes
\end{itemize}

\vspace{0.3cm}

\textbf{Panel Administrativo}
\begin{itemize}
    \item Gestión de productos

    \item Sistema de mensajes

\begin{frame}{Lecciones Aprendidas}
\begin{enumerate}
    \item \textbf{Valor de los Mocks:} Implementación de mocks para servicios externos permitió testing confiable e independiente

    \item \textbf{Testing Temprano:} Implementar tests desde etapas iniciales facilita detección temprana de problemas

    \item \textbf{Mantenimiento de Tests E2E:} Usar data-testid en lugar de clases CSS para mayor estabilidad

    \item \textbf{Cobertura vs Calidad:} 44.39\% de cobertura es adecuado al cubrir casos críticos - más valioso que cobertura alta superficial

    \item \textbf{Tests de Usabilidad:} Usuarios reales revelan problemas de UX que tests automatizados no detectan

    \item \textbf{Estrategia Híbrida:} Combinación de pruebas unitarias (backend) y E2E (frontend) proporciona cobertura integral
\end{enumerate}
\end{frame}
